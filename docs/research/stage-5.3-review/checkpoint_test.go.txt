func TestStage53ReviewClearNeedsOwnershipEvidence(t *testing.T) {
	f := recoverySetup(t)
	base := saveRun(t, f, "review-base")
	write(t, filepath.Join(f.root, "same.txt"), []byte("user-only edit\n"), 0600)
	captured := saveRun(t, f, "review-user")
	receipt, err := f.recovery.Clear(context.Background(), ClearRequest{CommandID: "review-clear", ExpectedRevision: revision(t, f.engine), CheckpointID: captured.CheckpointID, BaselineCheckpointID: base.CheckpointID, OwnedPaths: map[string][]string{"repo": {"same.txt"}}})
	got, _ := os.ReadFile(filepath.Join(f.root, "same.txt"))
	if string(got) != "user-only edit\n" {
		t.Fatalf("cleared user work without any execution result/proven ownership: %s err=%v", receipt.State, err)
	}
}
func TestStage53ReviewClearRejectsLiveWriter(t *testing.T) {
	f := recoverySetup(t)
	base := saveRun(t, f, "review-base")
	write(t, filepath.Join(f.root, "same.txt"), []byte("captured edit\n"), 0600)
	captured := saveRun(t, f, "review-after")
	if _, err := f.engine.DB.SQL.Exec("UPDATE runs SET state='active',writer_state='unconfirmed' WHERE id=?", f.prepared.RunID); err != nil {
		t.Fatal(err)
	}
	receipt, err := f.recovery.Clear(context.Background(), ClearRequest{CommandID: "review-clear", ExpectedRevision: revision(t, f.engine), CheckpointID: captured.CheckpointID, BaselineCheckpointID: base.CheckpointID, OwnedPaths: map[string][]string{"repo": {"same.txt"}}})
	if err == nil {
		t.Fatalf("clear accepted active/unconfirmed writer without coordinator claim: %s", receipt.State)
	}
}
func TestStage53ReviewRestoreStagedObjectAfterPrune(t *testing.T) {
	f := recoverySetup(t)
	base := saveRun(t, f, "review-base")
	write(t, filepath.Join(f.root, "same.txt"), []byte("unique staged content\n"), 0600)
	git(t, "-C", f.root, "add", "same.txt")
	oid := strings.TrimSpace(string(git(t, "-C", f.root, "rev-parse", ":same.txt")))
	target := saveRun(t, f, "review-target")
	if _, err := f.recovery.Clear(context.Background(), ClearRequest{CommandID: "review-clear", ExpectedRevision: revision(t, f.engine), CheckpointID: target.CheckpointID, BaselineCheckpointID: base.CheckpointID, OwnedPaths: map[string][]string{"repo": {"same.txt"}}}); err != nil {
		t.Fatal(err)
	}
	git(t, "-C", f.root, "prune", "--expire", "now")
	if err := exec.Command("git", "-C", f.root, "cat-file", "-e", oid).Run(); err == nil {
		t.Fatal("fixture staged object was not pruned")
	}
	destination := saveRun(t, f, "review-destination")
	receipt, err := f.recovery.Restore(context.Background(), RestoreRequest{CommandID: "review-restore", ExpectedRevision: revision(t, f.engine), CheckpointID: target.CheckpointID, BaselineCheckpointID: base.CheckpointID, DestinationCheckpointID: destination.CheckpointID})
	if err != nil {
		t.Fatal(err)
	}
	if err := exec.Command("git", "-C", f.root, "cat-file", "-e", oid).Run(); err != nil {
		t.Fatalf("reported %s with index referencing missing staged object %s", receipt.State, oid)
	}
}
func TestStage53ReviewPathnameTraversalGap(t *testing.T) {
	root := t.TempDir()
	outside := t.TempDir()
	write(t, filepath.Join(root, "dir", "file"), []byte("expected"), 0600)
	write(t, filepath.Join(outside, "file"), []byte("outside user data"), 0600)
	target, err := validateParent(root, "dir/file", true)
	if err != nil {
		t.Fatal(err)
	}
	if err := os.Rename(filepath.Join(root, "dir"), filepath.Join(root, "old-dir")); err != nil {
		t.Fatal(err)
	}
	if err := os.Symlink(outside, filepath.Join(root, "dir")); err != nil {
		t.Fatal(err)
	}
	err = atomicWrite(target, []byte("replacement"), 0600, false)
	got, _ := os.ReadFile(filepath.Join(outside, "file"))
	if string(got) != "outside user data" {
		t.Fatalf("validated parent replacement escaped recovery root: outside=%q err=%v", got, err)
	}
}
