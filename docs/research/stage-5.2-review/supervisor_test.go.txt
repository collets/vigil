type reviewInspectFailure struct {
	*FixtureDriver
	fail bool
}

func (d *reviewInspectFailure) Inspect(ctx context.Context, p PreparedRun) (Observation, error) {
	if d.fail {
		return Observation{}, errors.New("inspection unavailable")
	}
	return d.FixtureDriver.Inspect(ctx, p)
}
func TestReviewReplayAfterCrash(t *testing.T) {
	f := setupFixture(t)
	ctx := context.Background()
	d := &reviewInspectFailure{FixtureDriver: f.driver}
	r := Runner{Engine: f.engine, Driver: d, Fault: func(s string) error {
		if s == "after_runtime_create" {
			return errors.New("crash")
		}
		return nil
	}}
	if err := r.startSegment(ctx, f.prepared); err != nil {
		t.Fatal(err)
	}
	call := func() error { return d.Create(ctx, f.prepared) }
	if err := r.ensureDriverEffect(ctx, f.prepared, 1, "runtime_create", call); err == nil {
		t.Fatal("expected crash")
	}
	r.Fault = nil
	d.fail = true
	err := r.ensureDriverEffect(ctx, f.prepared, 1, "runtime_create", call)
	if d.Calls["create"] != 1 {
		t.Fatalf("replayed runtime create after crash with unavailable inspection: calls=%d, err=%v", d.Calls["create"], err)
	}
}
func TestReviewQuarantinedOwnerDispatch(t *testing.T) {
	f := setupFixture(t)
	if err := f.owner.Close(); err != nil {
		t.Fatal(err)
	}
	r := Runner{Engine: f.engine, Driver: f.driver}
	result, err := r.Run(context.Background(), f.prepared, f.reservation, "write result")
	if f.driver.Calls["submit"] > 0 {
		t.Fatalf("dispatched using quarantined owner's stale reservation: result=%s err=%v", result.Status, err)
	}
}

type reviewSlowSubmit struct{ *FixtureDriver }

func (d *reviewSlowSubmit) Submit(ctx context.Context, p PreparedRun, s string) (string, string, error) {
	time.Sleep(time.Duration(p.ActiveLimitMS+100) * time.Millisecond)
	return d.FixtureDriver.Submit(ctx, p, s)
}
func TestReviewActiveBudgetSubmit(t *testing.T) {
	f := setupFixture(t)
	r := Runner{Engine: f.engine, Driver: &reviewSlowSubmit{f.driver}}
	result, err := r.Run(context.Background(), f.prepared, f.reservation, "write result")
	if err == nil {
		t.Fatalf("completed beyond persisted 30000ms active limit after 30100ms submit: %s", result.Status)
	}
}
func TestReviewFixtureSymlinkEscape(t *testing.T) {
	f := setupFixture(t)
	victim := filepath.Join(t.TempDir(), "victim")
	if err := os.WriteFile(victim, []byte("preserve"), 0600); err != nil {
		t.Fatal(err)
	}
	if err := os.Symlink(victim, filepath.Join(f.driver.Root, ".git", "review-escape")); err != nil {
		t.Fatal(err)
	}
	f.driver.RelativePath = ".git/review-escape"
	r := Runner{Engine: f.engine, Driver: f.driver}
	_, runErr := r.Run(context.Background(), f.prepared, f.reservation, "write result")
	got, err := os.ReadFile(victim)
	if err != nil {
		t.Fatal(err)
	}
	if string(got) != "preserve" {
		t.Fatalf("fixture overwrote file outside repository before result rejection: contents=%q runErr=%v", got, runErr)
	}
}
func TestReviewHumanWaitCheckpoint(t *testing.T) {
	f := setupFixture(t)
	f.prepared.ActiveLimitMS = 50
	r := Runner{Engine: f.engine, Driver: f.driver}
	ctx := context.Background()
	if err := r.startSegment(ctx, f.prepared); err != nil {
		t.Fatal(err)
	}
	if err := r.BeginHumanWait(ctx, f.prepared, true); err != nil {
		t.Fatal(err)
	}
	time.Sleep(70 * time.Millisecond)
	if err := r.checkpointSegment(ctx, f.prepared); err != nil {
		t.Fatalf("proven human wait incorrectly exhausted active budget: %v", err)
	}
}
func TestReviewDeliveredSubmissionReplay(t *testing.T) {
	f := setupFixture(t)
	f.driver.RelativePath = "src/.keep"
	f.driver.Content = []byte("fixture\n")
	ctx := context.Background()
	d := &reviewInspectFailure{FixtureDriver: f.driver}
	r := Runner{Engine: f.engine, Driver: d, Fault: func(s string) error {
		if s == "after_terminal_observe" {
			return errors.New("crash")
		}
		return nil
	}}
	if _, err := r.Run(ctx, f.prepared, f.reservation, "write result"); err == nil {
		t.Fatal("expected crash")
	}
	if err := r.ReconcileOpenSegment(ctx, f.prepared, 0); err != nil {
		t.Fatal(err)
	}
	r.Fault = nil
	d.fail = true
	_, err := r.Run(ctx, f.prepared, f.reservation, "write result")
	if d.Calls["submit"] != 1 {
		t.Fatalf("delivered native prompt replayed when inspection unavailable: calls=%d err=%v", d.Calls["submit"], err)
	}
}
func TestReviewFixtureHardlinkEscape(t *testing.T) {
	f := setupFixture(t)
	victim := filepath.Join(t.TempDir(), "victim")
	if err := os.WriteFile(victim, []byte("fixture\n"), 0600); err != nil {
		t.Fatal(err)
	}
	target := filepath.Join(f.driver.Root, "src", ".keep")
	if err := os.Remove(target); err != nil {
		t.Fatal(err)
	}
	if err := os.Link(victim, target); err != nil {
		t.Fatal(err)
	}
	f.driver.RelativePath = "src/.keep"
	r := Runner{Engine: f.engine, Driver: f.driver}
	result, runErr := r.Run(context.Background(), f.prepared, f.reservation, "write result")
	got, err := os.ReadFile(victim)
	if err != nil {
		t.Fatal(err)
	}
	if string(got) != "fixture\n" {
		t.Fatalf("in-scope fixture overwrote external hardlink: contents=%q status=%s err=%v", got, result.Status, runErr)
	}
}

type reviewStopIntent struct {
	*FixtureDriver
	r *Runner
	t *testing.T
}

func (d *reviewStopIntent) Start(context.Context, PreparedRun) error {
	return errors.New("force containment")
}
func (d *reviewStopIntent) Stop(ctx context.Context, p PreparedRun) (Observation, error) {
	var n int
	if err := d.r.Engine.DB.SQL.QueryRow("SELECT count(*) FROM execution_effects WHERE generation_id=? AND kind='containment_stop'", p.GenerationID).Scan(&n); err != nil {
		d.t.Fatal(err)
	}
	if n == 0 {
		d.t.Error("Stop invoked without durable containment intent")
	}
	return d.FixtureDriver.Stop(ctx, p)
}
func TestReviewContainmentIntent(t *testing.T) {
	f := setupFixture(t)
	r := Runner{Engine: f.engine}
	r.Driver = &reviewStopIntent{f.driver, &r, t}
	_, _ = r.Run(context.Background(), f.prepared, f.reservation, "write result")
}
